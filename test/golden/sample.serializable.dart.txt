import 'package:json_annotation/json_annotation.dart';

part "root.g.dart";

@JsonSerializable(explicitToJson: true)
class Root {
  final int id;
  final String name;
  final double rating;
  final double price;
  final bool active;
  @JsonKey(name: "created_at")
  final DateTime createdAt;
  final List<DateTime> dates;
  @JsonKey(name: "class")
  final String classValue;
  final Address address;
  final List<String> tags;
  final List<List<int>> matrix;
  final List<dynamic> mixed;
  final List<Order> orders;

  const Root({
    required this.id,
    required this.name,
    required this.rating,
    required this.price,
    required this.active,
    required this.createdAt,
    required this.dates,
    required this.classValue,
    required this.address,
    required this.tags,
    required this.matrix,
    required this.mixed,
    required this.orders,
  });

  factory Root.fromJson(Map<String, dynamic> json) => _$RootFromJson(json);

  Map<String, dynamic> toJson() => _$RootToJson(this);

  static List<Root> parseList(dynamic list) {
    if (list is! List) {
      return [];
    }
    return list.map((e) => Root.fromJson(e as Map<String, dynamic>)).toList();
  }
}
