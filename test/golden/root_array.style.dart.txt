class Root {
  int id;
  String user_name;
  double? score;

  Root({
    required this.id,
    required this.user_name,
    this.score,
  });

  factory Root.fromJson(Map<String, dynamic> json) {
    return Root(
      id: json["id"] as int,
      user_name: json["user_name"] as String,
      score: (json["score"] as num?)?.toDouble(),
    );
  }

  Map<String, dynamic> toJson() {
    return {
      "id": id,
      "user_name": user_name,
      "score": score,
    };
  }

  static List<Root> parseList(dynamic list) {
    if (list is! List) {
      return [];
    }
    return list.map((e) => Root.fromJson(e as Map<String, dynamic>)).toList();
  }
}
