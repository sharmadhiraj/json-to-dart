class Root {
  final int id;
  final String name;
  final double rating;
  final double price;
  final bool active;
  final String createdAt;
  final List<String> dates;
  final String classValue;
  final Address address;
  final List<String> tags;
  final List<List<int>> matrix;
  final List<dynamic> mixed;
  final Map<String, dynamic> meta;
  final Map<String, Score> scores;
  final Map<String, List<int>> refs;
  final List<Order> orders;

  const Root({
    required this.id,
    required this.name,
    required this.rating,
    required this.price,
    required this.active,
    required this.createdAt,
    required this.dates,
    required this.classValue,
    required this.address,
    required this.tags,
    required this.matrix,
    required this.mixed,
    required this.meta,
    required this.scores,
    required this.refs,
    required this.orders,
  });

  factory Root.fromJson(Map<String, dynamic> json) {
    return Root(
      id: json["id"] as int,
      name: json["name"] as String,
      rating: (json["rating"] as num).toDouble(),
      price: (json["price"] as num).toDouble(),
      active: json["active"] as bool,
      createdAt: json["created_at"] as String,
      dates: (json["dates"] as List).map((e) => e as String).toList(),
      classValue: json["class"] as String,
      address: Address.fromJson(json["address"] as Map<String, dynamic>),
      tags: (json["tags"] as List).map((e) => e as String).toList(),
      matrix: (json["matrix"] as List).map((e) => (e as List).map((e2) => e2 as int).toList()).toList(),
      mixed: List<dynamic>.from(json["mixed"] as List),
      meta: Map<String, dynamic>.from(json["meta"] as Map),
      scores: (json["scores"] as Map<String, dynamic>).map((k, v) => MapEntry(k, Score.fromJson(v as Map<String, dynamic>))),
      refs: (json["refs"] as Map<String, dynamic>).map((k, v) => MapEntry(k, (v as List).map((e2) => e2 as int).toList())),
      orders: Order.parseList(json["orders"]),
    );
  }

  Map<String, dynamic> toJson() {
    return {
      "id": id,
      "name": name,
      "rating": rating,
      "price": price,
      "active": active,
      "created_at": createdAt,
      "dates": dates,
      "class": classValue,
      "address": address.toJson(),
      "tags": tags,
      "matrix": matrix,
      "mixed": mixed,
      "meta": meta,
      "scores": scores.map((k, v) => MapEntry(k, v.toJson())),
      "refs": refs,
      "orders": orders.map((e) => e.toJson()).toList(),
    };
  }

  static List<Root> parseList(dynamic list) {
    if (list is! List) {
      return [];
    }
    return list.map((e) => Root.fromJson(e as Map<String, dynamic>)).toList();
  }
}

class Address {
  final String city;
  final String postcode;

  const Address({
    required this.city,

  factory Address.fromJson(Map<String, dynamic> json) {
    return Address(
      city: json["city"] as String,

  }

    return {
      "city": city,
      "postcode": postcode,

  static List<Address> parseList(dynamic list) {
    if (list is! List) {
      return [];
    }
    return list.map((e) => Address.fromJson(e as Map<String, dynamic>)).toList();
  }
}

class Score {
  final int value;
  final List<String>? tags;

  const Score({
    required this.value,
    this.tags,
  });

  factory Score.fromJson(Map<String, dynamic> json) {
    return Score(
      value: json["value"] as int,
      tags: json["tags"] == null ? null : (json["tags"] as List).map((e) => e as String).toList(),
    );
  }

  Map<String, dynamic> toJson() {
    return {
      "value": value,
      "tags": tags,
    };
  }

  static List<Score> parseList(dynamic list) {
    if (list is! List) {
      return [];
    }
    return list.map((e) => Score.fromJson(e as Map<String, dynamic>)).toList();
  }
}

class Order {
  final int id;
  final double total;
  final dynamic note;
  final List<Line>? lines;
  final bool? extra;

  const Order({
    required this.id,
    required this.total,
    this.note,
    this.lines,
    this.extra,
  });

  factory Order.fromJson(Map<String, dynamic> json) {
    return Order(
      id: json["id"] as int,
      total: (json["total"] as num).toDouble(),
      note: json["note"],
      lines: json["lines"] == null ? null : Line.parseList(json["lines"]),
      extra: json["extra"] as bool?,
    );
  }

  Map<String, dynamic> toJson() {
    return {
      "id": id,
      "total": total,
      "note": note,
      "lines": lines?.map((e) => e.toJson()).toList(),
      "extra": extra,
    };
  }

  static List<Order> parseList(dynamic list) {
    if (list is! List) {
      return [];
    }
    return list.map((e) => Order.fromJson(e as Map<String, dynamic>)).toList();
  }
}
